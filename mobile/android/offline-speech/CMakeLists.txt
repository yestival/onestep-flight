# Build the independent eSpeak NG command-line program; the app communicates
# through standard UTF-8 input and WAV output, with no linked JNI library.
cmake_minimum_required(VERSION 3.22)
project(flight_offline_speech VERSION 1.52.0 LANGUAGES C)
if(NOT ESPEAK_SOURCE_DIR)
  message(FATAL_ERROR "Set ESPEAK_SOURCE_DIR to the verified eSpeak NG source")
endif()
include(CheckSymbolExists)
check_symbol_exists(mkstemp "stdlib.h" HAVE_MKSTEMP)
set(USE_ASYNC OFF)
set(USE_KLATT ON)
set(USE_LIBPCAUDIO OFF)
set(USE_LIBSONIC OFF)
set(USE_MBROLA OFF)
set(USE_SPEECHPLAYER OFF)
set(BUILD_SHARED_LIBS OFF)
add_library(espeak-include INTERFACE)
target_include_directories(espeak-include INTERFACE "${ESPEAK_SOURCE_DIR}/src/include" "${ESPEAK_SOURCE_DIR}/src/include/compat")
add_subdirectory("${ESPEAK_SOURCE_DIR}/src/ucd-tools" ucd)
add_subdirectory("${ESPEAK_SOURCE_DIR}/src/libespeak-ng" synthesizer)
add_executable(flight_espeak "${ESPEAK_SOURCE_DIR}/src/espeak-ng.c")
set_target_properties(flight_espeak PROPERTIES OUTPUT_NAME libflight_espeak.so)
target_link_libraries(flight_espeak PRIVATE espeak-ng espeak-ng-config)
target_link_options(flight_espeak PRIVATE "-Wl,-z,max-page-size=16384")
